cmake_minimum_required(VERSION 3.20)
project(CZ_Help LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 4)
  message(FATAL_ERROR "Build with Visual Studio 2022 -A Win32; GoldSrc requires x86.")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _CRT_SECURE_NO_WARNINGS)
add_compile_options(/utf-8 /W4 /permissive-)
add_library(cz_core STATIC shared/core.cpp)
target_include_directories(cz_core PUBLIC shared)
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
target_link_libraries(core_tests PRIVATE cz_core)
add_test(NAME core_behavior COMMAND core_tests)
add_library(cz_session STATIC CZ_Help/session.cpp)
target_include_directories(cz_session PUBLIC CZ_Help)
target_link_libraries(cz_session PUBLIC shell32)
add_executable(session_client_tests tests/session_client_tests.cpp)
target_link_libraries(session_client_tests PRIVATE cz_session)
add_test(NAME session_client COMMAND session_client_tests)
add_library(cz_game_detection STATIC CZ_Help/game_detection.cpp)
target_include_directories(cz_game_detection PUBLIC CZ_Help)
target_link_libraries(cz_game_detection PUBLIC cz_session wbemuuid ole32 oleaut32)
add_executable(game_detection_tests tests/game_detection_tests.cpp)
target_link_libraries(game_detection_tests PRIVATE cz_game_detection)
add_test(NAME game_detection COMMAND game_detection_tests)
add_executable(CZ_Help WIN32 CZ_Help/main.cpp CZ_Help/settings.cpp CZ_Help/version.rc)
target_link_libraries(CZ_Help PRIVATE cz_core cz_session cz_game_detection user32 gdi32 shell32)
add_test(NAME application_startup COMMAND CZ_Help --smoke-test)
add_test(NAME managed_session_ui COMMAND powershell.exe -NoLogo -NoProfile -ExecutionPolicy Bypass
  -File "${CMAKE_SOURCE_DIR}/tests/native_session_tests.ps1" -BinaryPath "$<TARGET_FILE:CZ_Help>" -VerifyModes)
add_test(NAME game_discovery_ui COMMAND powershell.exe -NoLogo -NoProfile -ExecutionPolicy Bypass
  -File "${CMAKE_SOURCE_DIR}/tests/game_discovery_ui_tests.ps1" -BinaryPath "$<TARGET_FILE:CZ_Help>" -FixtureBinary "$<TARGET_FILE:game_detection_tests>")
# These tests temporarily run the single-instance helper and share its INI.
set_tests_properties(managed_session_ui game_discovery_ui game_detection PROPERTIES RUN_SERIAL TRUE)
add_executable(channel_tests tests/channel_tests.cpp)
target_link_libraries(channel_tests PRIVATE cz_core)
add_test(NAME snapshot_channel COMMAND channel_tests)
add_executable(settings_tests tests/settings_tests.cpp CZ_Help/settings.cpp)
target_include_directories(settings_tests PRIVATE CZ_Help)
target_link_libraries(settings_tests PRIVATE cz_core)
add_test(NAME settings_persistence COMMAND settings_tests)
add_executable(render_control_tests tests/render_control_tests.cpp)
target_include_directories(render_control_tests PRIVATE shared)
add_test(NAME render_controls COMMAND render_control_tests)
add_executable(bhop_channel_tests tests/bhop_channel_tests.cpp)
target_include_directories(bhop_channel_tests PRIVATE shared)
add_test(NAME bhop_channel COMMAND bhop_channel_tests)
if(EXISTS "${CMAKE_SOURCE_DIR}/plugin/CMakeLists.txt")
  add_subdirectory(plugin)
endif()
